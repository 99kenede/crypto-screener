    <script>
        fetch('results.json?t=' + new Date().getTime())
            .then(response => response.json())
            .then(data => {
                document.getElementById('loading').style.display = 'none';
                document.getElementById('content').style.display = 'block';
                document.getElementById('meta').innerHTML = `Timeframe: <b>${data.timeframe}</b> | Última atualização: <b>${data.last_update}</b>`;
                document.getElementById('count-above').innerText = data.total_above;
                document.getElementById('count-below').innerText = data.total_below;

                const tbodyAbove = document.getElementById('tbody-above');
                data.above_zero.forEach(coin => {
                    tbodyAbove.innerHTML += `<tr><td><a href="${coin.tv_link}" target="_blank">${coin.symbol}</a></td><td>$${coin.price.toFixed(coin.price < 1 ? 4 : 2)}</td><td class="above"><b>${coin.cci}</b></td></tr>`;
                });

                const tbodyBelow = document.getElementById('tbody-below');
                data.below_zero.forEach(coin => {
                    tbodyBelow.innerHTML += `<tr><td><a href="${coin.tv_link}" target="_blank">${coin.symbol}</a></td><td>$${coin.price.toFixed(coin.price < 1 ? 4 : 2)}</td><td class="below"><b>${coin.cci}</b></td></tr>`;
                });
            })
            .catch(error => {
                document.getElementById('loading').innerText = 'Erro ao carregar. O GitHub Actions ainda não rodou?';
            });
    </script>
